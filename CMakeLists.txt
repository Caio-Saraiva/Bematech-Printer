cmake_minimum_required(VERSION 3.20)
project(BematechPrinter VERSION 0.1.1 LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
file(GLOB web_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/*")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/web_assets.h"
  COMMAND ${CMAKE_COMMAND} -DWEB_DIR=${CMAKE_CURRENT_SOURCE_DIR}/web -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/web_assets.h -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedWeb.cmake
  DEPENDS ${web_files} cmake/EmbedWeb.cmake VERBATIM)
add_library(bematech src/bematech.c)
target_include_directories(bematech PUBLIC include)
add_executable(bematech-printer src/main.c src/transport.c src/server.c "${CMAKE_CURRENT_BINARY_DIR}/web_assets.h")
target_include_directories(bematech-printer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(bematech-printer PRIVATE bematech)
if(WIN32)
  target_link_libraries(bematech-printer PRIVATE ws2_32 shell32)
endif()
if(MSVC)
  target_compile_definitions(bematech-printer PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(bematech PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(bematech-printer PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(core-tests tests/test_core.c)
  target_link_libraries(core-tests PRIVATE bematech)
  add_test(NAME core COMMAND core-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:bematech-printer>)
  endif()
endif()
install(TARGETS bematech-printer RUNTIME DESTINATION bin)
install(FILES README.md README.pt-BR.md LICENSE DESTINATION share/doc/bematech-printer)
install(DIRECTORY docs/ DESTINATION share/doc/bematech-printer/docs)
if(UNIX AND NOT APPLE)
  install(FILES packaging/bematech-printer.desktop DESTINATION share/applications)
  install(FILES web/icon.svg DESTINATION share/icons/hicolor/scalable/apps RENAME bematech-printer.svg)
endif()
set(CPACK_PACKAGE_NAME "bematech-printer")
set(CPACK_PACKAGE_VENDOR "Bematech Printer contributors")
set(CPACK_PACKAGE_CONTACT "https://github.com/Caio-Saraiva/Bematech-Printer/issues")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Local Bematech thermal printing studio")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Caio Saraiva")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_NSIS_DISPLAY_NAME "Bematech Printer")
set(CPACK_PACKAGE_EXECUTABLES "bematech-printer;Bematech Printer")
if(WIN32)
  set(CPACK_GENERATOR "ZIP;NSIS")
elseif(APPLE)
  set(CPACK_GENERATOR "TGZ")
else()
  set(CPACK_GENERATOR "TGZ;DEB")
endif()
include(CPack)
